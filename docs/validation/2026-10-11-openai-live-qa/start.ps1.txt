param()
$ErrorActionPreference='Stop'
$repo=Split-Path (Split-Path $PSScriptRoot -Parent) -Parent
Set-Location $repo
$configPath=Join-Path $PSScriptRoot 'config.yaml'
$configText=[IO.File]::ReadAllText($configPath)
if($configText -match 'api_key:\s*(""|'''')\s*(#.*)?(?:\r?\n|$)'){throw 'OpenAI API token is empty. Fill providers.openai.api_key in the local config first.'}
if(-not(Test-Path (Join-Path $PSScriptRoot 'BUILD-READY.txt'))){throw 'Application build is not ready. Wait for the current-source stand build.'}
docker compose -f (Join-Path $PSScriptRoot 'compose.yaml') up -d
if($LASTEXITCODE -ne 0){throw 'Owned stand startup failed.'}
for($i=0;$i -lt 30;$i++){try{$r=Invoke-WebRequest http://127.0.0.1:58280/readyz -SkipHttpErrorCheck; if($r.StatusCode -eq 200){Write-Output 'Stand ready at http://127.0.0.1:58280. No live inference has been submitted.';return}}catch{};Start-Sleep -Seconds 1}
throw 'Readiness did not become healthy. No live inference was submitted.'
