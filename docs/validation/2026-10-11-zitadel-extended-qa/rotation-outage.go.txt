package main

import (
	"context"
	"encoding/base64"
	"encoding/json"
	"errors"
	"fmt"
	"github.com/complynx/clx-scripter/internal/api"
	"github.com/complynx/clx-scripter/internal/config"
	"github.com/complynx/clx-scripter/internal/identity"
	"io"
	"net/http"
	"net/http/httptest"
	"net/url"
	"os"
	"os/exec"
	"strings"
	"time"
)

type client struct{ ClientID, ClientSecret string }
type oauth struct {
	AccessToken string `json:"access_token"`
}

func main() {
	const base = "http://localhost:18090"
	const dir = "management.local/zitadel-extended-qa/"
	var s struct {
		Human         struct{ UserID string }
		Project       struct{ ID string }
		App, OIDC     client
		Service, User oauth
		Scope         string
	}
	b, e := os.ReadFile(dir + "state.json")
	must(e)
	must(json.Unmarshal(b, &s))
	pb, e := os.ReadFile(dir + "data/admin.pat")
	must(e)
	pat := strings.TrimSpace(string(pb))
	failures := 0
	check := func(name string, ok bool) {
		if ok {
			fmt.Println("PASS " + name)
		} else {
			fmt.Println("FAIL " + name)
			failures++
		}
	}
	call := func(path string, body any) json.RawMessage {
		b, _ := json.Marshal(body)
		r, _ := http.NewRequest("POST", base+path, strings.NewReader(string(b)))
		r.Header.Set("Authorization", "Bearer "+pat)
		r.Header.Set("Content-Type", "application/json")
		resp, e := http.DefaultClient.Do(r)
		must(e)
		defer resp.Body.Close()
		if resp.StatusCode != 200 {
			panic(fmt.Sprintf("web-key API status=%d", resp.StatusCode))
		}
		raw, e := io.ReadAll(resp.Body)
		must(e)
		return raw
	}
	exchange := func() oauth {
		f := url.Values{"grant_type": {"urn:ietf:params:oauth:grant-type:token-exchange"}, "subject_token": {s.Human.UserID}, "subject_token_type": {"urn:zitadel:params:oauth:token-type:user_id"}, "actor_token": {s.Service.AccessToken}, "actor_token_type": {"urn:ietf:params:oauth:token-type:access_token"}, "requested_token_type": {"urn:ietf:params:oauth:token-type:jwt"}, "scope": {s.Scope}}
		r, _ := http.NewRequest("POST", base+"/oauth/v2/token", strings.NewReader(f.Encode()))
		r.SetBasicAuth(s.OIDC.ClientID, s.OIDC.ClientSecret)
		r.Header.Set("Content-Type", "application/x-www-form-urlencoded")
		resp, e := http.DefaultClient.Do(r)
		must(e)
		defer resp.Body.Close()
		if resp.StatusCode != 200 {
			panic(fmt.Sprintf("exchange status=%d", resp.StatusCode))
		}
		var t oauth
		must(json.NewDecoder(resp.Body).Decode(&t))
		return t
	}
	cfg := config.Identity{Issuer: base, Audience: s.Project.ID, ClientID: s.App.ClientID, ClientSecret: s.App.ClientSecret}
	newAuth := func() *identity.Authenticator { a, e := identity.New(cfg, nil, nil); must(e); return a }
	a := newAuth()
	ctx := context.Background()
	req := func(token string) *http.Request {
		r := httptest.NewRequest("GET", "http://harness/v1/chats/default", nil)
		r.Header.Set("Authorization", "Bearer "+token)
		return r
	}
	old := exchange()
	_, e = a.Authenticate(ctx, req(old.AccessToken))
	check("pre-rotation token warms discovery and JWKS cache", e == nil)
	var created struct{ ID string }
	must(json.Unmarshal(call("/v2beta/web_keys", map[string]any{}), &created))
	check("documented WebKeyService creates actual new signing key", created.ID != "")
	call("/v2beta/web_keys/"+created.ID+"/activate", map[string]any{})
	time.Sleep(time.Second)
	rotated := exchange()
	check("post-activation JWT kid changed to new real provider key", kid(rotated.AccessToken) == created.ID && kid(old.AccessToken) != created.ID)
	_, e = a.Authenticate(ctx, req(rotated.AccessToken))
	check("warmed authenticator refreshes JWKS for rotated unknown kid", e == nil)
	_, e = a.Authenticate(ctx, req(old.AccessToken))
	check("retained previous signing key validates already issued token", e == nil)
	cold := newAuth()
	_, e = cold.Authenticate(ctx, req(rotated.AccessToken))
	check("fresh authenticator accepts actual rotated signing key", e == nil)
	must(exec.Command("docker", "stop", "-t", "5", "clx-zitadel-extended-qa-id-20261011").Run())
	_, e = a.Authenticate(ctx, req(rotated.AccessToken))
	check("warmed discovery and JWKS cache cannot bypass issuer outage", errors.Is(e, identity.ErrUnavailable))
	_, e = newAuth().Authenticate(ctx, req(rotated.AccessToken))
	check("cold authenticator classifies stopped issuer unavailable", errors.Is(e, identity.ErrUnavailable))
	w := httptest.NewRecorder()
	api.New(nil, a, api.Options{}).ServeHTTP(w, req(rotated.AccessToken))
	check("HTTP returns 503 during outage despite cached valid signing key", w.Code == 503)
	fmt.Printf("RESULT failures=%d\n", failures)
	if failures > 0 {
		os.Exit(1)
	}
}
func kid(t string) string {
	var h struct{ Kid string }
	b, e := base64.RawURLEncoding.DecodeString(strings.Split(t, ".")[0])
	must(e)
	must(json.Unmarshal(b, &h))
	return h.Kid
}
func must(e error) {
	if e != nil {
		panic(e)
	}
}
