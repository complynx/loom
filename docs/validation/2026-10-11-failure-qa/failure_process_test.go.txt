type qaProcessProvider struct{ mode, marker string }
func (p qaProcessProvider) Respond(ctx context.Context, _ provider.Request) (provider.Response,error) {
 if p.mode=="model" { _=os.WriteFile(p.marker,[]byte("dispatch"),0600); <-ctx.Done(); return provider.Response{},ctx.Err() }

}
func TestQAChildWorker(t *testing.T) {
 if os.Getenv("CLX_QA_CHILD")!="1" { t.Skip("subprocess helper") }
 pool,err:=pgxpool.New(context.Background(),os.Getenv("CLX_QA_DSN")); require.NoError(t,err); defer pool.Close()
 opts:=recoveryOptions("killed-worker"); opts.Lease=300*time.Millisecond; opts.Heartbeat=50*time.Millisecond
 opts.Script=func(ctx context.Context,_ domain.Run,_ domain.Principal,_,_ string)(json.RawMessage,error){
  _,e:=pool.Exec(context.Background(),"INSERT INTO qa_effects(effect) VALUES ('service-write')"); if e!=nil{return nil,e}
  require.NoError(t,os.WriteFile(os.Getenv("CLX_QA_MARKER"),[]byte("write committed"),0600))
  <-ctx.Done(); return nil,ctx.Err()
 }
 require.NoError(t,runs.New(store.New(pool,nil),qaProcessProvider{os.Getenv("CLX_QA_MODE"),os.Getenv("CLX_QA_MARKER")},opts).Execute(context.Background(),os.Getenv("CLX_QA_RUN")))
}
func TestQARealWorkerKillBoundaries(t *testing.T) {
 for _,mode:=range []string{"model","tool"} { t.Run(mode,func(t *testing.T){
 f:=newFixture(t); _,err:=f.pool.Exec(context.Background(),"CREATE TABLE qa_effects(effect text)"); require.NoError(t,err)
 r:=f.createRun(t,runRequest("","","Exercise crash recovery"),"crash-"+mode)
 marker:=filepath.Join(t.TempDir(),"boundary.txt"); bin,err:=os.Executable(); require.NoError(t,err)
 cmd:=exec.Command(bin,"-test.run=^TestQAChildWorker$","-test.v")
 cmd.Env=append(os.Environ(),"CLX_QA_CHILD=1","CLX_QA_DSN="+f.pool.Config().ConnString(),"CLX_QA_RUN="+r.ID,"CLX_QA_MODE="+mode,"CLX_QA_MARKER="+marker)
 require.NoError(t,cmd.Start()); killed:=false; t.Cleanup(func(){if !killed{_=cmd.Process.Kill();_=cmd.Wait()}})
 deadline:=time.Now().Add(8*time.Second); for {if _,err=os.Stat(marker);err==nil{break}; require.True(t,time.Now().Before(deadline),"child did not reach %s boundary",mode);time.Sleep(10*time.Millisecond)}
 var pending int;require.NoError(t,f.pool.QueryRow(context.Background(),"SELECT count(*) FROM steps WHERE run_id=$1 AND state='pending'",r.ID).Scan(&pending));require.Equal(t,1,pending)
 require.NoError(t,cmd.Process.Kill());_=cmd.Wait();killed=true
 time.Sleep(400*time.Millisecond)
 recovery:=&recoveringProvider{}; opts:=recoveryOptions("replacement-worker");var duplicateEffects atomic.Int32
 opts.Script=func(context.Context,domain.Run,domain.Principal,string,string)(json.RawMessage,error){duplicateEffects.Add(1);return json.RawMessage(`{}`),nil}
 restartErr:=runs.New(f.repository,recovery,opts).Execute(context.Background(),r.ID)
 result:=f.run(t,r.ID);var effects,exposure int;require.NoError(t,f.pool.QueryRow(context.Background(),"SELECT count(*) FROM qa_effects").Scan(&effects));require.NoError(t,f.pool.QueryRow(context.Background(),"SELECT count(*) FROM platform_provider_attempts WHERE run_id=$1",r.ID).Scan(&exposure))
 t.Logf("actual kill mode=%s restart_err=%v state=%s reason=%s effects=%d duplicate_effects=%d replacement_inference=%d sponsored_unknown=%d spent=%d reserved=%d",mode,restartErr,result.State,result.Reason,effects,duplicateEffects.Load(),len(recovery.requests),exposure,result.SpentMicrocredits,result.ReservedMicrocredits)
 require.Zero(t,duplicateEffects.Load())
 if mode=="model"{require.NoError(t,restartErr);require.Equal(t,domain.Completed,result.State);require.Equal(t,1,exposure);require.EqualValues(t,15,result.SpentMicrocredits);require.Zero(t,result.ReservedMicrocredits)}else{require.EqualError(t,restartErr,"inference_uncertain");require.Equal(t,1,effects);require.Equal(t,domain.Reconciling,result.State);require.Empty(t,recovery.requests)}
 }) }
}

type qaInterruptedStore struct{*store.Store; armed *atomic.Bool; entered chan struct{}; hit atomic.Bool}
func(s *qaInterruptedStore)ExecutionRun(ctx context.Context,id string)(domain.Run,error){
 pc,_,_,_:=runtime.Caller(1); directLoop:=strings.HasSuffix(runtime.FuncForPC(pc).Name(),".(*Engine).executionLoop"); if directLoop&&s.armed.Load()&&s.hit.CompareAndSwap(false,true){close(s.entered);<-ctx.Done();return domain.Run{},ctx.Err()};return s.Store.ExecutionRun(ctx,id)
}
type qaArmProvider struct{armed *atomic.Bool}
func(p qaArmProvider)Respond(context.Context,provider.Request)(provider.Response,error){p.armed.Store(true);return provider.Response{ID:"late-final",Text:"Final must not overwrite cancel",Output:[]provider.Item{provider.Message("assistant","Final must not overwrite cancel")},Usage:provider.Usage{InputTokens:10,OutputTokens:5}},nil}
func TestQACancelAtDatabaseReadDiagnostic(t *testing.T){
 f:=newFixture(t); r:=f.createRun(t,runRequest("","","Cancel read boundary"),"cancel-db-read");var armed atomic.Bool
 s:=&qaInterruptedStore{Store:f.repository,armed:&armed,entered:make(chan struct{})};opts:=recoveryOptions("cancel-at-db-worker"); opts.SponsoredAttemptLimit=0
 engine:=runs.New(s,qaArmProvider{&armed},opts);done:=make(chan error,1);go func(){done<-engine.Execute(context.Background(),r.ID)}()
 select{case<-s.entered:case<-time.After(3*time.Second):t.Fatal("read boundary not reached")}
 code,raw:=f.request(t,"alice","POST","/v1/runs/"+r.ID+"/cancel","cancel-boundary",nil);require.Equal(t,200,code,string(raw))
 stopctx,cancel:=context.WithTimeout(context.Background(),time.Second);defer cancel();require.NoError(t,engine.Stop(stopctx));err:=<-done
 after:=f.run(t,r.ID);t.Logf("injected read cancellation err=%v state=%s cancel_requested=%t result=%q reason=%s spent=%d reserved=%d",err,after.State,after.CancelRequested,after.Result,after.Reason,after.SpentMicrocredits,after.ReservedMicrocredits)
 require.ErrorIs(t,err,context.Canceled);require.NotEqual(t,domain.Completed,after.State);require.Empty(t,after.Result);fmt.Println("Cancellation DB-load window reproduced; observe state above")
}
