$ErrorActionPreference='Stop'
$fixture='management.local/shutdown-fix/compose.yaml'
$evidence='docs/validation/2026-10-11-shutdown-fix'
$api='clx-shutdown-fix-20261011-api-1'
$pg='clx-shutdown-fix-20261011-postgres-1'
$results=@()
foreach($scenario in @('pg-stopped-repeat-1','pg-stopped-repeat-2')) {
 docker start $api | Out-Null
 $response=$null
 for($attempt=0;$attempt -lt 60;$attempt++) {
  try {$response=Invoke-WebRequest http://127.0.0.1:58480/readyz -SkipHttpErrorCheck -TimeoutSec 2; if($response.StatusCode -eq 200){break}} catch {}
  Start-Sleep -Milliseconds 100
 }
 if($response.StatusCode -ne 200){throw 'API not ready'}
 $runId=$null
 if($scenario -eq 'active-inference-healthy') {
  $headers=@{Authorization='Bearer qa-runtime-write';'Idempotency-Key'='shutdown-active-healthy-fixture'}
  $chat=Invoke-RestMethod http://127.0.0.1:58480/v1/chats -Method POST -Headers $headers -ContentType application/json -Body '{}'
  $body=@{chat_id=$chat.id;input=@{text='QA_CANCEL'};interaction=@{questions='text';approvals=$true};delivery=@{mode='async'};budget=@{max_microcredits=10000}} | ConvertTo-Json -Depth 10
  $run=Invoke-RestMethod http://127.0.0.1:58480/v1/runs -Method POST -Headers $headers -ContentType application/json -Body $body
  $runId=$run.run_id
  for($attempt=0;$attempt -lt 100;$attempt++) {
   $runState=Invoke-RestMethod "http://127.0.0.1:58480/v1/runs/$runId" -Headers $headers
   if($runState.status -eq 'running'){break}
   Start-Sleep -Milliseconds 50
  }
  $runState | ConvertTo-Json -Depth 25 | Set-Content "$evidence/healthy-active-before.json"
  if($runState.status -ne 'running'){throw 'Active run not running'}
 }
 if($scenario.StartsWith('pg-stopped')){docker stop $pg | Out-Null}
 $since=[DateTimeOffset]::UtcNow.ToUnixTimeSeconds()-1
 docker kill --signal TERM $api | Out-Null
 $exit=docker wait $api
 $until=[DateTimeOffset]::UtcNow.ToUnixTimeSeconds()+1
 $events=@(docker events --since $since --until $until --filter "container=$api" --format '{{json .}}')
 $events | Set-Content "$evidence/$scenario-events.jsonl"
 $parsed=@($events | ConvertFrom-Json)
 $kill=$parsed | Where-Object Action -eq 'kill' | Select-Object -Last 1
 $die=$parsed | Where-Object Action -eq 'die' | Select-Object -Last 1
 $elapsed=([decimal]$die.timeNano-[decimal]$kill.timeNano)/1000000
 $limit=if($scenario -eq 'active-inference-healthy'){1000}else{5000}
 $results+=@{scenario=$scenario;elapsed_ms=$elapsed;limit_ms=$limit;passed=($elapsed -lt $limit -and $exit -eq '0');exit_code=$exit;run_id=$runId}
 docker logs $api 2>&1 | Set-Content "$evidence/$scenario-api.log"
 docker inspect $api --format '{{json .State}}' | Set-Content "$evidence/$scenario-state.json"
 if($scenario.StartsWith('pg-stopped')){docker start $pg | Out-Null}
 Write-Output "$scenario $elapsed ms exit=$exit"
}
$results | ConvertTo-Json -Depth 10 | Set-Content "$evidence/repeated-results.json"
