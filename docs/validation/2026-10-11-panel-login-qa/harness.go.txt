package main

import (
 "context"
 "encoding/json"
 "fmt"
 "net/http"
 "net/url"
 "os"
 "strings"
 "github.com/jackc/pgx/v5/pgxpool"
 "github.com/complynx/clx-scripter/internal/api"
 "github.com/complynx/clx-scripter/internal/config"
 "github.com/complynx/clx-scripter/internal/domain"
 "github.com/complynx/clx-scripter/internal/identity"
 "github.com/complynx/clx-scripter/internal/store"
)
func main(){
 var s struct{Project struct{ID string}; App struct{ClientID,ClientSecret string};OIDC struct{ClientID string}}
 b,e:=os.ReadFile("management.local/panel-login-qa/state.json");must(e);must(json.Unmarshal(b,&s))
 env,e:=os.ReadFile("management.local/panel-login-qa/.env");must(e)
 password:="";for _,line:=range strings.Split(string(env),"\n"){if strings.HasPrefix(line,"QA_POSTGRES_PASSWORD="){password=strings.TrimSpace(strings.TrimPrefix(line,"QA_POSTGRES_PASSWORD="))}}
 dsn:="postgres://postgres:"+url.QueryEscape(password)+"@localhost:58339/panelqa?sslmode=disable"
 ctx:=context.Background();must(store.Migrate(ctx,dsn));pool,e:=pgxpool.New(ctx,dsn);must(e);defer pool.Close()
 auth,e:=identity.New(config.Identity{Issuer:"http://localhost:58390",Audience:s.Project.ID,ClientID:s.App.ClientID,ClientSecret:s.App.ClientSecret,RolePermissions:map[string][]string{"panel-reader":{"diagnostics.read"}},PermissionsCacheTTLSeconds:1,JWKSCacheTTLSeconds:300},nil,nil);must(e)
 login,e:=api.NewOperatorLogin("http://localhost:58390",s.Project.ID,config.Operator{ClientID:s.OIDC.ClientID,RedirectURI:"http://localhost:58380/admin"});must(e)
 handler:=api.New(store.New(pool,nil),safeAuth{auth},api.Options{OperatorLogin:login,DiagnosticsHealth:func()domain.DiagnosticsHealth{return domain.DiagnosticsHealth{Dependencies:map[string]bool{"database":true}}}})
 fmt.Println("QA panel http://localhost:58380/admin ; real isolated issuer/store ; provider execution absent")
 must(http.ListenAndServe("127.0.0.1:58380",handler))
}
func must(e error){if e!=nil {panic(e)}}

type safeAuth struct{a *identity.Authenticator}
func(s safeAuth)Authenticate(ctx context.Context,r *http.Request)(domain.Principal,error){p,e:=s.a.Authenticate(ctx,r);fmt.Printf("QA authentication path=%s success=%t org_present=%t diagnostics_read=%t\n",r.URL.Path,e==nil,p.OrgID!="",p.Has("diagnostics.read"));return p,e}
